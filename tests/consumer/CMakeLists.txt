cmake_minimum_required(VERSION 3.28)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(renderengine_consumer LANGUAGES C CXX)
  set(BUILD_TESTING OFF CACHE BOOL "Build renderer tests" FORCE)
  add_subdirectory(../.. renderengine)
endif()

add_executable(scene_consumer main.cpp)
target_link_libraries(scene_consumer PRIVATE renderengine::renderengine)
set_target_properties(scene_consumer PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(scene_consumer PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(scene_consumer PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
